def vista_con(cota):
    vista = VistaPlantaPorticos()
    vista.resize(600, 400)
    vista.set_datos(EJES, NIVELES, ESTRUCTURA)
    vista.set_losas(LOSAS)
    vista.set_portico_activo("Pórtico 1")
    vista.set_nivel_activo(cota)
    return vista
