1847|     def _encuadre(self, escena):
1848|         """Devuelve el encuadre de la planta: (px, py, xmin, xmax, ...).
1849| 
1850|         ``px`` y ``py`` son las funciones que convierten coordenadas de planta
1851|         (en metros) a píxeles del widget. Se separan del ``paintEvent`` para
1852|         poder reutilizar la misma transformación al colocar los rótulos.
1853|         """
1854|         xs = [p[0] for p in escena["puntos"]]
1855|         ys = [p[1] for p in escena["puntos"]]
1856|         if not xs or not ys:
1857|             for familia, items in self.ejes.items():
1858|                 for item in items:
1859|                     c = self._numero(item.get("coordenada_m"))
1860|                     if c is None:
1861|                         continue
1862|                     (xs if familia == "x" else ys).append(c)
1863|         if not xs or not ys:
1864|             xs = [0.0, 10.0]
1865|             ys = [0.0, 10.0]
1866| 
1867|         xmin, xmax = min(xs), max(xs)
1868|         ymin, ymax = min(ys), max(ys)
1869|         aire = max(xmax - xmin, ymax - ymin, 5.0) * 0.05 + 0.5
1870|         xmin -= aire
1871|         xmax += aire
1872|         ymin -= aire
1873|         ymax += aire
1874|         # Margenes para las burbujas de los ejes (van fuera del área de obra)
1875|         # y para la leyenda y el pie de texto.
1876|         margen_x, margen_y = 78.0, 62.0
1877|         rango_x = max(xmax - xmin, 3.0)
1878|         rango_y = max(ymax - ymin, 3.0)
1879|         # Zoom acotado para que la planta ocupe mejor el área disponible.
1880|         # Factor reducido para que la estructura respire y no quede amontanada.
1881|         escala = min(
1882|             max(self.width() - 2 * margen_x, 1.0) / rango_x,
1883|             max(self.height() - 2 * margen_y, 1.0) / rango_y,
1884|         ) * 0.92 * self.zoom_factor
1885|         ox = (self.width() - rango_x * escala) / 2
1886|         oy = (self.height() - rango_y * escala) / 2
1887| 
1888|         def px(x):
1889|             return ox + (x - xmin) * escala
1890| 
1891|         def py(y):
1892|             return oy + (ymax - y) * escala
1893| 
1894|         return px, py, xmin, xmax, ymin, ymax, escala
